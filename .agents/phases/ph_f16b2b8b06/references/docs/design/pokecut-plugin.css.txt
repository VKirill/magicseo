/*
 * Pokecut plugin kit: the brand layer for BB-сервис plugin screens.
 * Rules: docs/DESIGN.md. Reference implementation: plugins/bb-plugin-lane-pilot/app.css.
 *
 * Copy into the plugin's app.css after the Tailwind imports, then replace
 *   SLUG -> the plugin id from package.json (bb.id), e.g. lane-pilot
 *   xx-  -> a short plugin prefix, e.g. lp-
 * The prefix matters: these classes are not scoped by the build, so two
 * plugins with the same class name would style each other's screens.
 *
 * The --xx-* tokens read the Pokecut theme's --pk-* tokens and fall back to
 * BB's own palette, so the screen stays tidy under any BB theme, light or dark.
 * Radii: 8px rows, buttons, inputs; 6px segments in a 9px track; 12px inner
 * cards; 14px single cards and the header strip; 16px panels; 20px dialogs.
 * Rules sit in the components layer, so a Tailwind utility on an element wins.
 */

[data-bb-plugin="SLUG"],
[data-bb-plugin-root]:not([data-bb-plugin]) {
  --xx-well: var(--pk-well, color-mix(in oklab, var(--muted) 45%, var(--card)));
  --xx-card: var(--pk-card, var(--card));
  --xx-outline: var(--pk-outline, var(--input));
  --xx-hairline: var(--pk-hairline, var(--border));
  --xx-drop: var(--pk-drop, 0 1px 2px rgb(0 0 0 / 0.06));
  --xx-card-drop: var(--pk-card-drop, none);
  --xx-raise: 0 1px 3px 0 rgb(0 0 0 / 0.12);
  --xx-accent: var(--pk-accent-gradient, linear-gradient(var(--primary), var(--primary)));
  --xx-accent-edge: var(--pk-accent-edge, none);
  --xx-success: var(--success, #3fae4a);
  --xx-success-text: var(--diff-added, #2f8a45);
  --xx-info: var(--timeline-accent, #3e7be8);
  --xx-warning: var(--warning, #e8a33d);
  --xx-warning-text: var(--warning-text, #b4540f);
  --xx-danger-text: var(--destructive-text, var(--destructive));
}
